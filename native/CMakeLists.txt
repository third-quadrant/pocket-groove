cmake_minimum_required(VERSION 3.24)
project(pocketgroove_audio LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(GROOVE_CORE_ONLY "Only build deterministic mixer tests" OFF)
add_library(groove_mixer STATIC mixer.cpp)
target_include_directories(groove_mixer PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(mixer_test mixer_test.cpp)
target_link_libraries(mixer_test PRIVATE groove_mixer)
enable_testing()
add_test(NAME mixer COMMAND mixer_test)
if(NOT GROOVE_CORE_ONLY)
  find_package(JNI REQUIRED)
  include(FetchContent)
  set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "PortAudio legacy CMake compatibility")
  set(PA_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(PA_BUILD_STATIC ON CACHE BOOL "" FORCE)
  set(PA_USE_ASIO OFF CACHE BOOL "" FORCE)
  set(PA_USE_DS OFF CACHE BOOL "" FORCE)
  set(PA_USE_WMME OFF CACHE BOOL "" FORCE)
  set(PA_USE_WDMKS OFF CACHE BOOL "" FORCE)
  set(PA_USE_WASAPI ON CACHE BOOL "" FORCE)
  set(PA_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
  FetchContent_Declare(portaudio
    URL https://codeload.github.com/PortAudio/portaudio/zip/refs/tags/v19.7.0
    URL_HASH SHA256=ce1e7b27ad362cb9745de7da3d266a996b085ca75a12ac62c881f88ab6894acf)
  FetchContent_MakeAvailable(portaudio)
  add_library(pocketgroove_audio SHARED audio_jni.cpp)
  target_include_directories(pocketgroove_audio PRIVATE ${JNI_INCLUDE_DIRS})
  target_link_libraries(pocketgroove_audio PRIVATE groove_mixer portaudio_static)
  target_compile_definitions(pocketgroove_audio PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  set_target_properties(pocketgroove_audio PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
